        // Bio character counter
        const bioInput = document.getElementById('bio');
        const bioCounter = document.getElementById('bio-counter');
        
        bioInput.addEventListener('input', function() {
            const currentLength = this.value.length;
            bioCounter.textContent = `${currentLength}/280`;
            if(currentLength >= 280) {
                bioCounter.classList.add('text-error');
            } else {
                bioCounter.classList.remove('text-error');
            }
        });

        // Toast functionality
        function showToast(message, type = 'success') {
            const container = document.getElementById('toast-container');
            
            const toast = document.createElement('div');
            toast.className = `glass-panel rounded-lg px-4 py-3 shadow-lg flex items-center gap-3 toast-enter pointer-events-auto border-l-4 ${type === 'success' ? 'border-l-[#a855f7]' : 'border-l-error'}`;
            
            const icon = type === 'success' ? 'check_circle' : 'error';
            const iconColor = type === 'success' ? 'text-[#a855f7]' : 'text-error';
            
            toast.innerHTML = `
                <span class="material-symbols-outlined ${iconColor}">${icon}</span>
                <span class="font-label-md text-label-md text-on-surface">${message}</span>
                <button onclick="this.parentElement.remove()" class="ml-auto text-on-surface-variant hover:text-on-surface">
                    <span class="material-symbols-outlined text-sm">close</span>
                </button>
            `;
            
            container.appendChild(toast);
            
            // Remove after 3 seconds
            setTimeout(() => {
                toast.style.opacity = '0';
                toast.style.transform = 'translateY(10px)';
                toast.style.transition = 'all 0.3s ease';
                setTimeout(() => toast.remove(), 300);
            }, 3000);
        }
    </script>

        function handleSend() {
            const messageEl = document.getElementById('message');
            const errorEl = document.getElementById('messageError');
            
            if (!messageEl.value.trim()) {
                errorEl.classList.remove('hidden');
                messageEl.parentElement.classList.add('border-error');
                messageEl.parentElement.classList.remove('border-outline-variant/30');
                return;
            }
            
            errorEl.classList.add('hidden');
            messageEl.parentElement.classList.remove('border-error');
            messageEl.parentElement.classList.add('border-outline-variant/30');

            const name = document.getElementById('name').value;
            const subject = document.getElementById('subject').value;
            const message = messageEl.value;

            let mailtoLink = `mailto:sobhagyaverma16@gmail.com?subject=${encodeURIComponent(subject || 'CodeIT Contact Form')}&body=${encodeURIComponent(message)}`;
            if (name) {
                mailtoLink += encodeURIComponent(`\n\nFrom: ${name}`);
            }

            window.location.href = mailtoLink;
        }

        // Clear error on input
        document.getElementById('message').addEventListener('input', function() {
            document.getElementById('messageError').classList.add('hidden');
            this.parentElement.classList.remove('border-error');
            this.parentElement.classList.add('border-outline-variant/30');
        });
